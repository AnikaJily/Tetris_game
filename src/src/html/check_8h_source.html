1484/* String comparison macros with improved output compared to ck_assert() */
1485/* OP might be any operator that can be used in '0 OP strcmp(X,Y)' comparison.
1486 */
1487/* String pointer could be compared againts NULL with == (NULLEQ = 1) and !=
1488 * (NULLNE = 1) operators. */
1489/* The x and y parameter swap in strcmp() is needed to handle >, >=, <, <=
1490 * operators. */
1491/* If the x or y parameter is NULL its value will be printed without quotes. */
1492#define _ck_assert_str(X, OP, Y, NULLEQ, NULLNE) \
1493 do { \
1494 const char *_ck_x = (X); \
1495 const char *_ck_y = (Y); \
1496 const char *_ck_x_s; \
1497 const char *_ck_y_s; \
1498 const char *_ck_x_q; \
1499 const char *_ck_y_q; \
1500 if (_ck_x != NULL) { \
1501 _ck_x_q = "\""; \
1502 _ck_x_s = _ck_x; \
1503 } else { \
1504 _ck_x_q = ""; \
1505 _ck_x_s = "(null)"; \
1506 } \
1507 if (_ck_y != NULL) { \
1508 _ck_y_q = "\""; \
1509 _ck_y_s = _ck_y; \
1510 } else { \
1511 _ck_y_q = ""; \
1512 _ck_y_s = "(null)"; \
1513 } \
1514 ck_assert_msg((NULLEQ && (_ck_x == NULL) && (_ck_y == NULL)) || \
1515 (NULLNE && ((_ck_x == NULL) || (_ck_y == NULL)) && \
